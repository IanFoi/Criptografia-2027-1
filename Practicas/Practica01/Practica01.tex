% Actualmente: Verde Olivo OliveGreen
% Usar para mantener el mismo color de resaltado de texto a secciones de

\newcommand{\textColor}[1]{
    % Si estamos en un entorno matematico
    \ifmmode
        % No aplicamos negritas
        \textcolor{OliveGreen}{#1}
    % En un entorno de texto limpio usamos textbf
    \else
        \textbf{\textcolor{OliveGreen}{#1}}
    \fi
}

% Copia de itemize para utilizar el color establecido en \textColor
% en las listas de viñetas que sea necesario.
\newlist{itemizeColor}{itemize}{3} % El 3 es la profundidad recomendada
                                   % para no perder la formalidad del doc 

% Configuracion global de itemizeColor
% Listas de viñetas del color predeterminado, asegurarse de que sea el
% mismo color del comando \textColor para que el documento se mantenga
% equilibrado. (Actuamente: OliveGreen)
% "font=" aplica el estilo a la viñeta
% \bfseries para aplicar negritas dentro del formato matematico
\setlist[itemizeColor]{font=\color{OliveGreen}\bfseries}

% Asignación de símbolos para cada nivel de itemizeColor
% Nivel 1: Punto (\textbullet)
\setlist[itemizeColor,1]{label=\textbullet}
% Nivel 2: Guion (\textendash)
\setlist[itemizeColor,2]{label=\textendash}
% Nivel 3: Asterisco (\textasteriskcentered)
\setlist[itemizeColor,3]{label=\textasteriskcentered}


% ======================================================================
% Enumeraciones (enumerate)
% ======================================================================

% Copia de enumerate para utilizar el color establecido en \textColor
% en las enumeraciones que sean necesarias.
\newlist{enumerateColor}{enumerate}{3} % El 3 es la profundidad recomendada
                                       % para no perder la formalidad del doc 

% Configuracion global de enumerateColor
% Enumeracion con el color predeterminado, se debe asegurar de que sea el
% mismo color del comando \textColor para que el documento se mantenga
% equilibrado. (Actuamente: OliveGreen)
% "font=" aplica el estilo a la enumeracion
% \bfseries para aplicar negritas dentro del formato matematico
\setlist[enumerateColor]{font=\color{OliveGreen}\bfseries}

% Asignación de numeraciones para cada nivel de enumerateColor
% Nivel 1: Numeros arabigos con punto 1. 2. 3. ...
\setlist[enumerateColor,1]{label=\arabic*.}
% Nivel 2: letras minusculas con parentesis a), b), c) ... z)
\setlist[enumerateColor,2]{label=\alph*)}
% Nivel 3: numeros romanos con punto i. ii. iii. ...
\setlist[enumerateColor,3]{label=\roman*.}
